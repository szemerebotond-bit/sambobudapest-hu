function doPost(e) {
  let p = (e && e.parameter) || {};
  // a chatbot JSON-t küld text/plain-ként
  if (e && e.postData && e.postData.contents && /^\s*\{/.test(e.postData.contents)) {
    try { p = Object.assign({}, p, JSON.parse(e.postData.contents)); } catch (err) {}
  }
  if (p.action === "chat") return json(handleChat(p));
  if (p.website) return json({ ok: true });            // spam-csapda
  const r = normalizeBooking(p);
  if (!r.name || !r.phone) return json({ ok: false, error: "missing" });
  saveBooking(r);
  return json({ ok: true });
}

function doGet() { return json({ ok: true, service: "sambo-foglalas" }); }

function normalizeBooking(p) {
  return {
    name: clean(p.name, 80), phone: clean(p.phone, 40), email: clean(p.email, 120),
    plan: clean(p.plan, 40) || "Csoportos", slot: clean(p.slot, 90),
    callback: /^(igen|true|yes)$/i.test(String(p.callback || "")) ? "Igen" : "Nem",
    style: clean(p.style, 40), level: clean(p.level, 40), message: clean(p.message, 1500)
  };
}

function saveBooking(r) {
  const lock = LockService.getScriptLock();
  lock.waitLock(15000);
  let url = "";
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    url = ss.getUrl();
    ss.getSheetByName(SHEET_NAME).appendRow([new Date(), "Új", safe(r.name), safe(r.phone), safe(r.email),
      safe(r.plan), safe(r.slot), r.callback, safe(r.style), safe(r.level), safe(r.message), ""]);
  } finally {
    lock.releaseLock();
  }
  try { notifyCoach(r, url); } catch (err) { console.error(err); }
  try { confirmApplicant(r); } catch (err) { console.error(err); }
}

function upcomingGroupSessions(n) {
  const days = ["vasárnap", "hétfő", "kedd", "szerda", "csütörtök", "péntek", "szombat"];
  const out = [];
  const now = new Date();
  const hour = Number(Utilities.formatDate(now, "Europe/Budapest", "H"));
  const d = new Date(now.getTime());
  if (hour >= 18) d.setDate(d.getDate() + 1);
  while (out.length < n) {
    const wd = Number(Utilities.formatDate(d, "Europe/Budapest", "u")) % 7; // 1=hétfő … 7=vasárnap → 0
    if (wd === 1 || wd === 3 || wd === 5) {
      out.push(days[wd] + " " + Utilities.formatDate(d, "Europe/Budapest", "yyyy.MM.dd.") + " 19:00–21:00");
    }
    d.setDate(d.getDate() + 1);
  }
  return out;
}

function handleChat(p) {
  const fallback = "Most nem tudok válaszolni, bocsi. Hívd Valentint: " + PHONE + ", vagy foglalj az oldalon lévő űrlappal.";
  const key = PropertiesService.getScriptProperties().getProperty("GEMINI_API_KEY");
  if (!key) return { ok: false, reply: fallback, error: "no_key" };

  // egyszerű védelem a túlhasználat ellen
  const cache = CacheService.getScriptCache();
  const bucket = "rate_" + Math.floor(Date.now() / 60000);
  const count = Number(cache.get(bucket) || 0);
  if (count > 40) return { ok: false, reply: "Most sokan írnak, próbáld újra egy perc múlva, vagy hívd: " + PHONE };
  cache.put(bucket, String(count + 1), 120);

  let msgs = [];
  try { msgs = Array.isArray(p.messages) ? p.messages : JSON.parse(p.messages || "[]"); } catch (err) {}
  msgs = msgs.slice(-20).filter(m => m && (m.role === "user" || m.role === "assistant") && m.text)
             .map(m => ({ role: m.role === "user" ? "user" : "model", parts: [{ text: String(m.text).slice(0, 1500) }] }));
  if (!msgs.length || msgs[msgs.length - 1].role !== "user") return { ok: false, reply: fallback };

  const base = { systemInstruction: { parts: [{ text: systemPrompt() }] }, tools: [BOOKING_TOOL],
                 generationConfig: { temperature: 0.4, maxOutputTokens: 600 } };
  let res = callGemini(key, Object.assign({ contents: msgs }, base));
  if (!res) return { ok: false, reply: fallback };

  const parts = (((res.candidates || [])[0] || {}).content || {}).parts || [];
  const call = parts.map(x => x.functionCall).filter(Boolean)[0];
  if (call && call.name === "create_booking") {
    const a = call.args || {};
    const r = normalizeBooking({
      name: a.name, phone: a.phone, email: a.email, plan: a.plan, slot: a.slot || "egyeztetjük",
      callback: a.callback ? "Igen" : "Nem", style: a.style, level: a.level,
      message: "[Chatbot] " + (a.message || "")
    });
    if (!r.name || !r.phone) return { ok: true, reply: "Ehhez még kell a neved és a telefonszámod." };
    saveBooking(r);
    const follow = msgs.concat([
      { role: "model", parts: [{ functionCall: call }] },
      { role: "user", parts: [{ functionResponse: { name: "create_booking", response: { ok: true, saved: true } } }] }
    ]);
    const res2 = callGemini(key, Object.assign({ contents: follow }, base));
    const t2 = textOf(res2) || ("Rögzítettem a foglalásod, " + r.name + "! Valentin hamarosan keres a megadott számon." +
                                (r.email ? " A visszaigazolást elküldtük e-mailben." : ""));
    return { ok: true, reply: t2, booked: true };
  }
  return { ok: true, reply: textOf(res) || fallback };
}

function callGemini(key, payload) {
  for (const model of GEMINI_MODELS) {
    const url = "https://generativelanguage.googleapis.com/v1beta/models/" + model + ":generateContent";
    const resp = UrlFetchApp.fetch(url, {
      method: "post", contentType: "application/json", muteHttpExceptions: true,
      headers: { "x-goog-api-key": key }, payload: JSON.stringify(payload)
    });
    const code = resp.getResponseCode();
    if (code === 200) return JSON.parse(resp.getContentText());
    console.error("Gemini " + model + " " + code + ": " + resp.getContentText().slice(0, 300));
    if (code !== 404 && code !== 400) break;
  }
  return null;
}

function textOf(res) {
  if (!res) return "";
  const parts = (((res.candidates || [])[0] || {}).content || {}).parts || [];
  return parts.map(x => x.text || "").join("").trim();
}

function clean(v, n) { return String(v == null ? "" : v).replace(/\s+/g, " ").trim().slice(0, n); }
function safe(v) { return /^[=+\-@]/.test(v) ? "'" + v : v; }
function json(o) {
  return ContentService.createTextOutput(JSON.stringify(o)).setMimeType(ContentService.MimeType.JSON);
}
